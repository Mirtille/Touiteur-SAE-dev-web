-- Script PLSQl :

-- Mettre à jour le nombre de touites pour un utilisateur : 


DELIMITER //

CREATE TRIGGER trg_increment_nb_touites
AFTER INSERT ON Touite
FOR EACH ROW
BEGIN
    UPDATE Utilisateur
    SET nb_touites = nb_touites + 1
    WHERE id_user = NEW.id_auteur;
END //

DELIMITER ;

-- Empêcher un utilisateur de se suivre : 


DELIMITER //

CREATE TRIGGER trg_verif_suivi
BEFORE INSERT ON UserSuivi
FOR EACH ROW
BEGIN
    DECLARE email_suiveur VARCHAR(128);
    DECLARE email_suivi VARCHAR(128);

    IF NEW.id_suiveur = NEW.id_suivi THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Un utilisateur ne peut pas se suivre lui-meme';
    END IF;

    SELECT email_user INTO email_suiveur
    FROM Utilisateur
    WHERE id_user = NEW.id_suiveur;

    SELECT email_user INTO email_suivi
    FROM Utilisateur
    WHERE id_user = NEW.id_suivi;

    IF LOWER(TRIM(email_suiveur)) = LOWER(TRIM(email_suivi)) THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Un utilisateur ne peut pas suivre quelqu''un ayant le meme email.';
    END IF;
END //

DELIMITER ;

-- Empêcher la modification de l'auteur d'un touite : 

DELIMITER //

CREATE TRIGGER trg_verif_auteur_touite
BEFORE UPDATE ON Touite
FOR EACH ROW
BEGIN
    IF OLD.id_auteur <> NEW.id_auteur THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Impossible de modifier l''auteur d''un touite';
    END IF;
END //

DELIMITER ;


-- Limiter un utilisateur à un maximum de 50 touites par jour : 

DELIMITER //

CREATE TRIGGER trg_limite_touites_jour
BEFORE INSERT ON Touite
FOR EACH ROW
BEGIN
    DECLARE nb_touites_jour INT;

    SELECT COUNT(*) INTO nb_touites_jour
    FROM Touite
    WHERE id_auteur = NEW.id_auteur
      AND DATE(date_publi) = DATE(NEW.date_publi);

    IF nb_touites_jour >= 50 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Limite de 50 touites par jour atteinte pour cet utilisateur';
    END IF;
END //

DELIMITER ;
